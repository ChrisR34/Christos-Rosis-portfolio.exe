public void Register(string cmd, Handler h)
  => _handlers.Add(cmd, h);
public string Route(string line)
{
  var (cmd, arg) = Parse(line);
  if (_handlers.TryGetValue(cmd, out var h))
    return h(arg);
  // unknown commands get an error, never a crash
  return $"ERR:UNKNOWN {cmd}";
}
private static (string, string) Parse(string line)
{
  var parts = line.Trim().Split(' ', 2);
  var cmd = parts.ElementAt(0).ToUpperInvariant();
  var arg = parts.Length > 1 ? parts.ElementAt(1) : "";
  return (cmd, arg);
}
